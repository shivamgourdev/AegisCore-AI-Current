0001: from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
0002: from pathlib import Path
0003: from urllib.error import HTTPError, URLError
0004: from urllib.request import Request, urlopen
0005: import os
0006: 
0007: ROOT = Path(__file__).resolve().parent
0008: BACKEND = "http://127.0.0.1:8000"
0009: 
0010: class Handler(SimpleHTTPRequestHandler):
0011:     def __init__(self, *args, **kwargs):
0012:         super().__init__(*args, directory=str(ROOT), **kwargs)
0013: 
0014:     def log_message(self, fmt, *args):
0015:         print("[sandbox]", fmt % args)
0016: 
0017:     def _proxy(self):
0018:         path = self.path
0019:         if not path.startswith("/api/"):
0020:             self.send_error(404, "Unknown API route")
0021:             return
0022:         target = BACKEND + path[4:]
0023:         length = int(self.headers.get("Content-Length", "0"))
0024:         body = self.rfile.read(length) if length else None
0025:         headers = {}
0026:         for key in ("Content-Type", "Accept"):
0027:             if self.headers.get(key):
0028:                 headers[key] = self.headers[key]
0029:         request = Request(target, data=body, headers=headers, method=self.command)
0030:         try:
0031:             with urlopen(request, timeout=120) as response:
0032:                 content = response.read()
0033:                 status = response.status
0034:                 content_type = response.headers.get("Content-Type", "application/json")
0035:         except HTTPError as exc:
0036:             content = exc.read()
0037:             status = exc.code
0038:             content_type = exc.headers.get("Content-Type", "application/json")
0039:         except (URLError, TimeoutError, OSError) as exc:
0040:             content = ('{"detail":"Cannot reach backend at ' + BACKEND +
0041:                        '. Start the FastAPI backend first. Details: ' +
0042:                        str(exc).replace('"', "'") + '"}').encode("utf-8")
0043:             status = 502
0044:             content_type = "application/json"
0045:         self.send_response(status)
0046:         self.send_header("Content-Type", content_type)
0047:         self.send_header("Content-Length", str(len(content)))
0048:         self.send_header("Cache-Control", "no-store")
0049:         self.end_headers()
0050:         self.wfile.write(content)
0051: 
0052:     def do_GET(self):
0053:         if self.path.startswith("/api/"):
0054:             return self._proxy()
0055:         if self.path == "/":
0056:             self.path = "/index.html"
0057:         return super().do_GET()
0058: 
0059:     def do_POST(self):
0060:         return self._proxy()
0061: 
0062: if __name__ == "__main__":
0063:     os.chdir(ROOT)
0064:     print("AegisCore AI Sandbox: http://127.0.0.1:5500")
0065:     print("Backend proxy target: " + BACKEND)
0066:     print("Keep this terminal open while using the sandbox.")

0032:     if not filename or not filename.strip():
0033:         raise DocumentExtractionError("A filename is required.")

0035:     if not content:
0036:         raise DocumentExtractionError("The document is empty.")
0037: 
0038:     extension = Path(filename).suffix.lower()
0039: 
0040:     if extension in {".txt", ".md"}:
0041:         try:
0042:             text = content.decode("utf-8-sig", errors="strict")
0043:         except UnicodeDecodeError as exc:

0045:                 "Text document must use valid UTF-8 encoding."
0046:             ) from exc

0048:         if "\x00" in text:
0049:             raise DocumentExtractionError("Text contains invalid NUL characters.")

0051:         if not text.strip():
0052:             raise DocumentExtractionError("No readable text was found.")
0053: 
0054:         return ExtractionResult(

0067:         raise DocumentExtractionError("The file does not have a valid PDF signature.")

0082:                     )

0085:                 raise DocumentExtractionError(
0086:                     "No selectable text found. This may be a scanned or image-only PDF; OCR is not available in this phase."
0087:                 )
0088: 
0089:             return ExtractionResult(
0090:                 filename=Path(filename).name,
0091:                 file_type="pdf",
0092:                 pages=tuple(extracted_pages),
